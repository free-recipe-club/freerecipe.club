Actually, since `loadRecipes` returns `Recipe[]` without slugs, and we need the slug for each variant, add a variant-aware loading function. Simpler approach: add a `loadRecipesWithSlugs()` function that returns `{ recipe: Recipe, slug: string }[]`, OR update `findVariants` to use `listRecipeSlugs` + `loadRecipe`:

```typescript
export function findVariants(parentSlug: string, recipesDir: string = RECIPES_DIR): { recipe: Recipe; slug: string }[] {
  let slugs = listRecipeSlugs(recipesDir)
  let results: { recipe: Recipe; slug: string }[] = []
  for (let slug of slugs) {
    let recipe = loadRecipe(getRecipeFilename(slug), recipesDir)
    if (recipe.variant_of === parentSlug) {
      results.push({ recipe, slug })
    }
  }
  return results
}
```

```typescript
// Step 3: Annotation quality checks
for (let group of recipe.components) {
  for (let i = 1; i < group.length; i++) {
    let item = group[i]
    if (typeof item !== 'string' && typeof item === 'object' && item !== null && 'annotations' in item) {
      let annItem = item as { text: string; annotations: { text: string; type: string; contributor: string }[] }
      for (let ann of annItem.annotations) {
        // Annotation text should not be empty after trim
        if (ann.text.trim().length === 0) {
          errors.push(`${file}: annotation on "${annItem.text}" has empty text`)
        }
        // Annotation type must be substitution or tip
        if (ann.type !== 'substitution' && ann.type !== 'tip') {
          errors.push(`${file}: annotation on "${annItem.text}" has invalid type "${ann.type}" (must be substitution or tip)`)
        }
        // Contributor should not be empty after trim
        if (ann.contributor.trim().length === 0) {
          errors.push(`${file}: annotation on "${annItem.text}" has empty contributor`)
        }
      }
    }
  }
}

// Same checks for direction annotations
for (let entry of recipe.directions) {
  let items: unknown[] = typeof entry === 'string' ? [] : Array.isArray(entry) ? entry.slice(1) : [entry]
  for (let item of items) {
    if (typeof item !== 'string' && typeof item === 'object' && item !== null && 'annotations' in item) {
      let annItem = item as { text: string; annotations: { text: string; type: string; contributor: string }[] }
      for (let ann of annItem.annotations) {
        if (ann.text.trim().length === 0) {
          errors.push(`${file}: annotation on direction "${annItem.text}" has empty text`)
        }
        if (ann.type !== 'substitution' && ann.type !== 'tip') {
          errors.push(`${file}: annotation on direction "${annItem.text}" has invalid type "${ann.type}"`)
        }
        if (ann.contributor.trim().length === 0) {
          errors.push(`${file}: annotation on direction "${annItem.text}" has empty contributor`)
        }
      }
    }
  }
}

// Variant reference check
if (recipe.variant_of) {
  let parentFile = recipe.variant_of.replace(/-/g, '_') + '.yml'
  let parentPath = path.join(RECIPES_DIR, parentFile)
  if (!fs.existsSync(parentPath)) {
    warnings.push(`${file}: variant_of references "${recipe.variant_of}" but ${parentFile} not found`)
  }
}
```
